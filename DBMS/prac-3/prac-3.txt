    o.odate,
    s.snum,

    o.amt = (
        SELECT MAX(o2.amt)
        FROM orders o2
        WHERE o2.odate = o.odate

CREATE VIEW sp8 AS
SELECT
    s.snum,
    s.sname
FROM
    salespeople s
JOIN
    orders o ON s.snum = o.snum
JOIN
    customer c ON o.cnum = c.cnum
WHERE
    o.amt = (
        SELECT MAX(o2.amt)
        FROM orders o2
        WHERE o2.odate = o.odate
    )
GROUP BY
    s.snum, s.sname
HAVING
    COUNT(o.onum) >= 3;

view created.
