-- Minimum Number from given 2 numbers

SQL> DECLARE
num1 NUMBER := 10;
num2 NUMBER := 20;
min_value NUMBER;
PROCEDURE find_min(num1 IN NUMBER, num2 IN NUMBER, min OUT NUMBER) IS
BEGIN
    IF num1 < num2 THEN
        min := num1;
    ELSE
        min := num2;
    END IF;
END find_min;
BEGIN
    find_min(num1, num2, min_value);
    DBMS_OUTPUT.PUT_LINE('The minimum value is: ' || min_value);
END;

The minimum value is: 10

PL/SQL procedure successfully completed.
